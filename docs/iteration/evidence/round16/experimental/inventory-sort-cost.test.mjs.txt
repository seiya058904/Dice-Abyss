import test from 'node:test';
import assert from 'node:assert/strict';
import {newGame,makeItem,itemValue} from '../src/engine.js';
import {selectInventoryItems} from '../src/inventory-query.js';

function fixture(count=1000) {
  const state=newGame(16161);state.floor=140;state.player.level=80;
  while(state.player.items.length<count){const index=state.player.items.length;state.player.items.push(makeItem(state,['weapon','armor','relic'][index%3],index%8));}
  return state.player.items;
}

// Independent baseline comparator: this is the shipped ordering contract,
// not a score-cache implementation. Cost counters are never timing samples.
function reference(items,ui={}) {
  const query=typeof ui.inventoryQuery==='string'?ui.inventoryQuery.trim().toLowerCase():'';
  return items.filter(item=>(!ui.inventorySlot||item.slot===ui.inventorySlot)
    &&(ui.inventoryRarity==null||ui.inventoryRarity===''||item.rarity===Number(ui.inventoryRarity))
    &&(!query||item.name.toLowerCase().includes(query)))
    .toSorted((a,b)=>ui.inventorySort==='rarity'?b.rarity-a.rarity||itemValue(b)-itemValue(a)
      :ui.inventorySort==='level'?b.level-a.level||itemValue(b)-itemValue(a):itemValue(b)-itemValue(a));
}

function valueReads(items,run) {
  const unique=[...new Set(items)],reads=new Map(unique.map(item=>[item,0]));
  const descriptors=unique.map(item=>Object.getOwnPropertyDescriptor(item,'enhance'));
  try{
    for(const [index,item]of unique.entries()){
      const amount=item.enhance;
      Object.defineProperty(item,'enhance',{enumerable:true,configurable:true,get(){reads.set(item,reads.get(item)+1);return amount;}});
    }
    return {result:run(),reads,total:[...reads.values()].reduce((sum,n)=>sum+n,0)};
  }finally{
    for(const [index,item]of unique.entries())if(descriptors[index])Object.defineProperty(item,'enhance',descriptors[index]);else delete item.enhance;
  }
}

test('ten-thousand-item ordering reuses values within one selection and keeps original objects',()=>{
  const items=fixture(10000),before=JSON.stringify(items);
  for(const inventorySort of ['power','rarity','level','unknown',undefined]){
    const ui={inventorySort},expected=reference(items,ui),measured=valueReads(items,()=>selectInventoryItems(items,ui));
    assert.deepEqual(measured.result,expected);
    for(const [index,item]of measured.result.entries())assert.strictEqual(item,expected[index]);
    assert(measured.total<=items.length);assert([...measured.reads.values()].every(n=>n<=1));
  }
  assert.equal(JSON.stringify(items),before);
});

test('empty and one-result selections do not evaluate unused sort values',()=>{
  const item=fixture(3)[0];
  for(const inventorySort of ['power','rarity','level','unknown']){
    assert.deepEqual(selectInventoryItems([],{inventorySort}),[]);
    const measured=valueReads([item],()=>selectInventoryItems([item],{inventorySort}));
    assert.deepEqual(measured.result,[item]);assert.equal(measured.total,0);
    const absent=valueReads([item],()=>selectInventoryItems([item],{inventorySort,inventoryQuery:'no matching item'}));
    assert.deepEqual(absent.result,[]);assert.equal(absent.total,0);
  }
});

test('primary-sort values that never tie require no item-power evaluation',()=>{
  const items=fixture(1000);for(const [index,item]of items.entries())item.level=index+1;
  const level=valueReads(items,()=>selectInventoryItems(items,{inventorySort:'level'}));
  assert.equal(level.total,0);assert.deepEqual(level.result,reference(items,{inventorySort:'level'}));
  const rarities=items.slice(0,8);for(const [index,item]of rarities.entries())item.rarity=index;
  const rarity=valueReads(rarities,()=>selectInventoryItems(rarities,{inventorySort:'rarity'}));
  assert.equal(rarity.total,0);assert.deepEqual(rarity.result,reference(rarities,{inventorySort:'rarity'}));
});

test('all tied values preserve inventory order, colliding IDs and repeated object references',()=>{
  const original=fixture(3)[0],items=Array.from({length:90},(_,index)=>({...original,id:index%2?'i42':original.id,name:`stable-${index}`}));
  items.splice(20,0,items[3]);items.splice(60,0,items[3]);
  for(const inventorySort of ['power','rarity','level','unknown']){
    const result=selectInventoryItems(items,{inventorySort});assert.notStrictEqual(result,items);
    for(const [index,item]of result.entries())assert.strictEqual(item,items[index]);
  }
});

test('filtering covers the full inventory before values and leaves excluded objects untouched',()=>{
  const items=fixture(10000);items[9999].name='尾部 Echo [Blade]';items[9998].name='尾部 ECHO [Blade]';
  items[9998].slot=items[9999].slot;items[9998].rarity=items[9999].rarity;
  const ui={inventoryQuery:'  echo [blade] ',inventorySlot:items[9999].slot,inventoryRarity:String(items[9999].rarity),inventorySort:'power'};
  const measured=valueReads(items,()=>selectInventoryItems(items,ui));
  assert.deepEqual(measured.result,reference(items,ui));assert.equal(measured.result.length,2);
  assert.equal(measured.total,2);assert.equal(measured.reads.get(items[0]),0);
  assert.equal(ui.inventoryQuery,'  echo [blade] ');
});

test('the next query sees in-place upgrades and replacement objects without stale cached values',()=>{
  const items=fixture(3);for(const item of items){item.rarity=0;item.level=10;item.levelXp=0;item.enhance=0;}
  for(const inventorySort of ['power','rarity','level']){
    selectInventoryItems(items,{inventorySort});
    items[0].level+=10;items[1].enhance+=1;items[2].levelXp+=1;
    assert.deepEqual(selectInventoryItems(items,{inventorySort}),reference(items,{inventorySort}));
    items[2]={...items[2],level:items[0].level+5};
    const next=selectInventoryItems(items,{inventorySort});assert.deepEqual(next,reference(items,{inventorySort}));
    assert(next.includes(items[2]));
  }
});

test('legal sparse optional and legacy fields retain score and reference ordering',()=>{
  const items=fixture(60);
  for(const [index,item]of items.entries()){
    delete item.locked;
    if(index%2){item.legacy=true;delete item.catalogId;delete item.profile;delete item.theme;}
    item.auditMetadata={unknown:true,index};
  }
  const before=JSON.stringify(items);
  for(const inventorySort of ['power','rarity','level',null]){
    const expected=reference(items,{inventorySort}),actual=selectInventoryItems(items,{inventorySort});
    assert.deepEqual(actual,expected);for(const [index,item]of actual.entries())assert.strictEqual(item,expected[index]);
  }
  assert.equal(JSON.stringify(items),before);
});
