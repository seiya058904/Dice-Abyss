import {itemValue} from './equipment.js';

// Search is a transient view choice. It never changes the owned item array,
// its order, a transaction's targets, or a saved field.
export const inventoryQuery = value => typeof value === 'string' ? value.trim() : '';

export function selectInventoryItems(items, ui = {}) {
  const query = inventoryQuery(ui.inventoryQuery).toLowerCase();
  const selected = items.filter(item => (!ui.inventorySlot || item.slot === ui.inventorySlot)
    && (ui.inventoryRarity == null || ui.inventoryRarity === '' || item.rarity === Number(ui.inventoryRarity))
    && (!query || item.name.toLowerCase().includes(query)));
  if (selected.length < 2) return selected;

  // Scores belong to this selection only: upgrading or replacing an item is
  // visible on the next call. Keep rows separate from item IDs and return the
  // original objects, including stable ties and any repeated references.
  const primary = ui.inventorySort === 'rarity' ? 'rarity' : ui.inventorySort === 'level' ? 'level' : null;
  const ranked = selected.map(item => ({item, value: primary ? undefined : itemValue(item)}));
  const value = entry => entry.value ??= itemValue(entry.item);
  // Named sorts need a score only when their primary values tie. This avoids
  // doing otherwise-unused work for inventories with all distinct levels.
  ranked.sort((a, b) => primary ? b.item[primary] - a.item[primary] || value(b) - value(a) : b.value - a.value);
  return ranked.map(entry => entry.item);
}
