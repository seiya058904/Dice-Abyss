import assert from 'node:assert/strict';
import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import * as E from '../baseline/Dice-Abyss/src/engine.js';
import * as B from '../baseline/Dice-Abyss/src/storage.js';
import * as A from './set-reuse-storage.mjs';
import * as C from './inventory-index-storage.mjs';
import * as D from './guarded-set-storage.mjs';
const sha=x=>createHash('sha256').update(typeof x==='string'?x:JSON.stringify(x)).digest('hex');
const variants=[['baseline',B],['set-reuse',A],['inventory-index',C],['guarded-set',D]];
const summary=values=>{const s=[...values].sort((a,b)=>a-b);return{n:s.length,min:s[0],p50:(s[s.length/2-1]+s[s.length/2])/2,p95:s[Math.ceil(s.length*.95)-1],max:s.at(-1),mean:s.reduce((a,b)=>a+b,0)/s.length};};
function fixture(count,position){
 const s=E.newGame(27017);s.runId='validator-audit-27017';s.floor=200;s.player.level=100;s.player.gold=1000000;
 while(s.player.items.length<count)s.player.items.push(E.makeItem(s,undefined,s.nextId%8));
 const list=slot=>s.player.items.filter(item=>item.slot===slot),pick=xs=>position==='first'?xs[0]:xs.at(-1);
 s.player.equipped.weapon=pick(list('weapon')).id;s.player.equipped.armor=pick(list('armor')).id;
 const relics=list('relic');s.player.equipped.relic=pick(relics).id;s.player.equipped.relic2=(position==='first'?relics[1]:relics.at(-2)).id;s.player.equipped.relic3=(position==='first'?relics[2]:relics.at(-3)).id;
 s.player.hp=E.stats(s).maxHp;s.auditMetadata={text:'Unknown fields 😀 界',nested:[{value:7}]};return s;
}
const report={kind:'Preliminary exact validator candidates',at:new Date().toISOString(),node:process.version,concurrentLoad:'Root baseline deep replay CPU activity; not formal performance evidence',sourceHashes:Object.fromEntries([['baseline','../baseline/Dice-Abyss/src/storage.js'],['set-reuse','./set-reuse-storage.mjs'],['inventory-index','./inventory-index-storage.mjs'],['guarded-set','./guarded-set-storage.mjs']].map(([key,p])=>[key,sha(readFileSync(new URL(p,import.meta.url),'utf8'))])),scriptSha256:sha(readFileSync(new URL(import.meta.url),'utf8')),cells:[]};
for(const count of [32,1000,10000])for(const position of ['first','last']){
 const state=fixture(count,position),before=JSON.stringify(state),raw=B.encodeSave(state),samples=Object.fromEntries(variants.map(([name])=>[name,[]])),order=[];
 for(let n=-8;n<32;n++){
  const turn=((n%4)+4)%4,ordered=variants.slice(turn).concat(variants.slice(0,turn));
  if(n>=0)order.push(ordered.map(([name])=>name));
  for(const [name,S]of ordered){const t=performance.now(),valid=S.validateSave(state),ms=performance.now()-t;assert.equal(valid,true);if(n>=0)samples[name].push(ms);}
 }
 assert.equal(JSON.stringify(state),before);for(const [,S]of variants)assert.equal(S.encodeSave(state),raw);
 report.cells.push({count,position,equipped:state.player.equipped,rawBytes:Buffer.byteLength(raw),stateSha256:sha(before),order,samplesMs:samples,summaryMs:Object.fromEntries(Object.entries(samples).map(([key,x])=>[key,summary(x)]))});
}
report.completedAt=new Date().toISOString();writeFileSync(new URL('./validator-guarded-preliminary.json',import.meta.url),JSON.stringify(report,null,2)+'\n');
